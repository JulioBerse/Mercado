import os
import psycopg2

def get_db_connection():
    # Pega a URL do banco das variáveis de ambiente ou usa a tua string Neon
    DATABASE_URL = os.environ.get('DATABASE_URL', 'SUA_STRING_CONEXAO_NEON_AQUI')
    
    # O Neon exige sslmode=require
    conn = psycopg2.connect(DATABASE_URL, sslmode='require')
    return conn
